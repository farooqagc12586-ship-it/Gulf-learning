from reportlab.lib.pagesizes import inch from reportlab.lib import colors from reportlab.lib.styles import getSampleStyleSheet, ParagraphStyle from reportlab.platypus import SimpleDocTemplate, Paragraph, Spacer, Table, TableStyle, PageBreak # ------------------------------------------------------------- # 1. ٹیگ کا سائز (Standard Tag Dimensions: 3.5 x 6.0 inches) # ------------------------------------------------------------- PAGE_WIDTH = 3.5 * inch PAGE_HEIGHT = 6.0 * inch def generate_fire_tag(filename="Fire_Extinguisher_Tag.pdf", hotline="059-1910990", location="Building A"): doc = SimpleDocTemplate( filename, pagesize=(PAGE_WIDTH, PAGE_HEIGHT), leftMargin=0.15 * inch, rightMargin=0.15 * inch, topMargin=0.15 * inch, bottomMargin=0.15 * inch ) story = [] styles = getSampleStyleSheet() # اسٹائلز (Styles) title_style = ParagraphStyle( 'TagTitle', parent=styles['Heading1'], fontSize=10, leading=12, textColor=colors.HexColor('#D32F2F'), # Red Theme alignment=1, # Center fontName='Helvetica-Bold' ) sub_style = ParagraphStyle( 'TagSub', fontSize=7, leading=9, fontName='Helvetica-Bold' ) pass_text_style = ParagraphStyle( 'PassText', fontSize=6.5, leading=8 ) # ============================================================= # SIDE 1: FRONT (HOW TO USE - P.A.S.S. INSTRUCTIONS) # ============================================================= story.append(Paragraph("HOW TO USE FIRE EXTINGUISHER", title_style)) story.append(Spacer(1, 4)) # لوکیشن اور ہاٹ لائن ٹیبل header_data = [ [Paragraph(f"LOCATION: {location}", sub_style)], [Paragraph(f"HOT LINE 24/7: {hotline}", sub_style)] ] header_table = Table(header_data, colWidths=[3.2 * inch]) header_table.setStyle(TableStyle([ ('BACKGROUND', (0,0), (-1,-1), colors.HexColor('#FFF9C4')), # Light Yellow Header ('ALIGN', (0,0), (-1,-1), 'CENTER'), ('BOX', (0,0), (-1,-1), 1, colors.HexColor('#FBC02D')), ('TOPPADDING', (0,0), (-1,-1), 2), ('BOTTOMPADDING', (0,0), (-1,-1), 2), ])) story.append(header_table) story.append(Spacer(1, 5)) # P.A.S.S Steps Table pass_data = [ [Paragraph("P", styles['Normal']), Paragraph("PULL THE 'PIN' IN THE HANDLE", pass_text_style)], [Paragraph("A", styles['Normal']), Paragraph("AIM THE 'NOZZLE' AT BASE OF FIRE", pass_text_style)], [Paragraph("S", styles['Normal']), Paragraph("SQUEEZE THE LEVER SLOWLY", pass_text_style)], [Paragraph("S", styles['Normal']), Paragraph("SWEEP FROM SIDE TO SIDE", pass_text_style)] ] pass_table = Table(pass_data, colWidths=[0.4 * inch, 2.8 * inch]) pass_table.setStyle(TableStyle([ ('VALIGN', (0,0), (-1,-1), 'MIDDLE'), ('GRID', (0,0), (-1,-1), 0.5, colors.HexColor('#FFCDD2')), ('BACKGROUND', (0,0), (0,-1), colors.HexColor('#FFEBEE')), ('TOPPADDING', (0,0), (-1,-1), 3), ('BOTTOMPADDING', (0,0), (-1,-1), 3), ])) story.append(pass_table) # اگلی سائیڈ پر جانے کے لیے Page Break story.append(PageBreak()) # ============================================================= # SIDE 2: BACK (INSPECTION TAG GRID) # ============================================================= story.append(Paragraph("FIRE EXTINGUISHER
INSPECTION TAG", title_style)) story.append(Spacer(1, 3)) meta_data = [ [Paragraph("Extinguisher No: ____________", sub_style)], [Paragraph("Type of Fire Extinguisher: ______", sub_style)] ] meta_table = Table(meta_data, colWidths=[3.2 * inch]) meta_table.setStyle(TableStyle([ ('TOPPADDING', (0,0), (-1,-1), 1), ('BOTTOMPADDING', (0,0), (-1,-1), 1) ])) story.append(meta_table) story.append(Spacer(1, 4)) # 12 ماہ کا معائنہ گریڈ (Inspection Table) grid_data = [["MONTH", "YEAR", "INSPECTED BY", "REMARKS"]] months = ["JAN.", "FEB.", "MAR.", "APRIL", "MAY", "JUNE", "JULY", "AUG.", "SEPT.", "OCT.", "NOV.", "DEC."] for m in months: grid_data.append([m, "", "", ""]) grid_table = Table(grid_data, colWidths=[0.6 * inch, 0.6 * inch, 1.1 * inch, 0.9 * inch]) grid_table.setStyle(TableStyle([ ('BACKGROUND', (0,0), (-1,0), colors.HexColor('#D32F2F')), # Red Header for Inspection ('TEXTCOLOR', (0,0), (-1,0), colors.white), ('ALIGN', (0,0), (-1,-1), 'CENTER'), ('FONTNAME', (0,0), (-1,0), 'Helvetica-Bold'), ('FONTSIZE', (0,0), (-1,-1), 5.5), ('GRID', (0,0), (-1,-1), 0.5, colors.black), ('TOPPADDING', (0,0), (-1,-1), 1.5), ('BOTTOMPADDING', (0,0), (-1,-1), 1.5), ])) story.append(grid_table) story.append(Spacer(1, 4)) warning_style = ParagraphStyle('Warn', fontSize=6.5, textColor=colors.red, alignment=1, fontName='Helvetica-Bold') story.append(Paragraph("DO NOT REMOVE THIS TAG", warning_style)) # پی ڈی ایف جنیریٹ کریں doc.build(story) print(f"File '{filename}' generated successfully!") # فنکشن کال کریں if __name__ == "__main__": generate_fire_tag()
