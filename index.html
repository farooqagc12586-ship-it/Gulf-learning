from meta_data = [ [import streamlit as st from reportlab.lib.pagesizes import inch from reportlab.lib import colors from reportlab.lib.styles import getSampleStyleSheet, ParagraphStyle from reportlab.platypus import SimpleDocTemplate, Paragraph, Spacer, Table, TableStyle, PageBreak import io st.set_page_config(page_title="HSE Tag Generator", layout="centered") st.title("🔥 Fire Extinguisher Tag Generator") # ویب پیج پر یوزر سے معلومات لینے کے لیے ان پٹ باکس location = st.text_input("Location / سائٹ کا نام", "Building A - Floor 1") hotline = st.text_input("Hotline Number", "059-1910990") # پی ڈی ایف جنریٹ کرنے کا فنکشن def create_pdf_bytes(loc, phone): buffer = io.BytesIO() doc = SimpleDocTemplate( buffer, pagesize=(3.5 * inch, 6.0 * inch), leftMargin=0.15 * inch, rightMargin=0.15 * inch, topMargin=0.15 * inch, bottomMargin=0.15 * inch ) story = [] styles = getSampleStyleSheet() title_style = ParagraphStyle('T', parent=styles['Heading1'], fontSize=10, textColor=colors.HexColor('#D32F2F'), alignment=1) sub_style = ParagraphStyle('S', fontSize=7, fontName='Helvetica-Bold') # Front Side story.append(Paragraph("HOW TO USE FIRE EXTINGUISHER", title_style)) story.append(Spacer(1, 5)) header_data = [[Paragraph(f"LOCATION: {loc}", sub_style)], [Paragraph(f"HOTLINE: {phone}", sub_style)]] t = Table(header_data, colWidths=[3.2 * inch]) t.setStyle(TableStyle([('BACKGROUND', (0,0), (-1,-1), colors.HexColor('#FFF9C4')), ('ALIGN', (0,0), (-1,-1), 'CENTER'), ('BOX', (0,0), (-1,-1), 1, colors.HexColor('#FBC02D'))])) story.append(t) story.append(PageBreak()) # Back Side story.append(Paragraph("FIRE EXTINGUISHER INSPECTION TAG", title_style)) story.append(Spacer(1, 5)) grid_data = [["MONTH", "YEAR", "INSPECTED BY", "REMARKS"]] for m in ["JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV", "DEC"]: grid_data.append([m, "", "", ""]) gt = Table(grid_data, colWidths=[0.6 * inch, 0.6 * inch, 1.1 * inch, 0.9 * inch]) gt.setStyle(TableStyle([('BACKGROUND', (0,0), (-1,0), colors.HexColor('#D32F2F')), ('TEXTCOLOR', (0,0), (-1,0), colors.white), ('GRID', (0,0), (-1,-1), 0.5, colors.black)])) story.append(gt) doc.build(story) buffer.seek(0) return buffer # ویب بٹن if st.button("Generate Tag PDF"): pdf_data = create_pdf_bytes(location, hotline) # پی ڈی ایف ڈاؤن لوڈ بٹن st.download_button( label="📥 Download Generated PDF Tag", data=pdf_data, file_name="Fire_Tag.pdf", mime="application/pdf" ) st.success("PDF کامیابی سے تیار ہو گئی ہے!")
